# 2. Функція для отримання кластеризованих похибок
get_cluster_se <- function(model, cluster_var) {
  vcovCL(model, cluster = cluster_var)
}

# 3. Функція для таблиці результатів з кластеризованими похибками
extract_clustered_info <- function(model, cluster_var) {
  # Кластеризована ков. матриця
  cl_vcov <- vcovCL(model, cluster = cluster_var)
  
  # Коефіцієнти + p-значення
  coef_test <- coeftest(model, vcov. = cl_vcov)
  
  # Логітні коефіцієнти
  logit_coef <- coef(model)
  
  # OR та CI на основі логітів
  OR <- exp(logit_coef)
  CI <- exp(confint.default(model, vcov. = cl_vcov))
  
  data.frame(
    Term = rownames(coef_test),
    OddsRatio = round(OR, 3),
    CI_low = round(CI[, 1], 3),
    CI_high = round(CI[, 2], 3),
    p_value = round(coef_test[, 4], 4)
  )
}

# 4. Отримання результатів
results1 <- extract_clustered_info(model1, ~ patient_nbr)
results2 <- extract_clustered_info(model2, ~ patient_nbr)
results3 <- extract_clustered_info(model3, ~ patient_nbr)

# 5. Збереження у CSV (або просто перегляд)

write.csv(results2, "model2_clustered.csv", row.names = FALSE)
